import json, os, shutil
from datetime import datetime
from pathlib import Path
from .user_space import UserSpace, AgentMemory, MemoryFact

class MemoryStorage:
    def __init__(self, memory_root: str = "memory"):
        self.memory_root = Path(memory_root)
        self.agents_dir = self.memory_root / "agents"
        self._ensure_dirs()

    def _ensure_dirs(self):
        self.memory_root.mkdir(parents=True, exist_ok=True)
        self.agents_dir.mkdir(exist_ok=True)

    def _atomic_write_json(self, path: Path, data: dict):
        tmp = path.with_suffix(".tmp")
        try:
            with open(tmp, "w", encoding="utf-8") as f:
                json.dump(data, f, ensure_ascii=False, indent=2)
            tmp.replace(path)  # 原子替换
        except Exception:
            tmp.unlink(missing_ok=True)
            raise

    def _atomic_read_json(self, path: Path) -> dict:
        if not path.exists():
            return {}
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)

    def load_user_space(self) -> UserSpace:
        data = self._atomic_read_json(self.memory_root / "user_space.json")
        if not data:
            return UserSpace()
        us = UserSpace()
        for k, v in data.get("profile", {}).items():
            if hasattr(us.profile, k):
                setattr(us.profile, k, v)
        for k, v in data.get("context", {}).items():
            if hasattr(us.context, k):
                setattr(us.context, k, v)
        us.facts = [self._dict_to_fact(f) for f in data.get("facts", [])]
        us.events = [self._dict_to_event(e) for e in data.get("events", [])]
        us.distilled_summary = data.get("distilled_summary", "")
        us.last_distilled_at = data.get("last_distilled_at", "")
        us.version = data.get("version", 1)
        return us

    def save_user_space(self, us: UserSpace):
        data = {
            "profile": self._to_dict(us.profile),
            "context": self._to_dict(us.context),
            "facts": [self._to_dict(f) for f in us.facts],
            "events": [self._to_dict(e) for e in us.events],
            "distilled_summary": us.distilled_summary,
            "last_distilled_at": us.last_distilled_at,
            "version": us.version,
            "updated_at": datetime.now().isoformat(),
        }
        self._atomic_write_json(self.memory_root / "user_space.json", data)
        self._auto_backup("user_space.json")

    def load_agent_memory(self, agent_id: str) -> AgentMemory:
        data = self._atomic_read_json(self.agents_dir / f"{agent_id}.json")
        if not data:
            return AgentMemory(agent_id=agent_id)
        am = AgentMemory(agent_id=agent_id)
        am.facts = [self._dict_to_fact(f) for f in data.get("facts", [])]
        am.events = [self._dict_to_event(e) for e in data.get("events", [])]
        am.working_memory = data.get("working_memory", {})
        am.domain_summary = data.get("domain_summary", "")
        am.agent_preferences = data.get("agent_preferences", {})
        return am

    def save_agent_memory(self, am: AgentMemory):
        data = {
            "agent_id": am.agent_id,
            "facts": [self._to_dict(f) for f in am.facts],
            "events": [self._to_dict(e) for e in am.events],
            "working_memory": am.working_memory,
            "domain_summary": am.domain_summary,
            "agent_preferences": am.agent_preferences,
            "version": getattr(am, "version", 1),
            "updated_at": datetime.now().isoformat(),
        }
        self._atomic_write_json(self.agents_dir / f"{am.agent_id}.json", data)

    def _to_dict(self, obj):
        from dataclasses import asdict
        return asdict(obj)

    def _dict_to_fact(self, d):
        return MemoryFact(**{k: v for k, v in d.items() if k in MemoryFact.__dataclass_fields__})

    def _dict_to_event(self, d):
        from .user_space import EpisodicEvent
        return EpisodicEvent(**{k: v for k, v in d.items() if k in EpisodicEvent.__dataclass_fields__})

    def _auto_backup(self, filename, max_backups=5):
        backup_dir = self.memory_root / "backups"
        backup_dir.mkdir(exist_ok=True)
        src = self.memory_root / filename
        if not src.exists():
            return
        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
        shutil.copy2(src, backup_dir / f"{filename}.{ts}")
        backups = sorted(backup_dir.glob(f"{filename}.*"))
        for old in backups[:-max_backups]:
            old.unlink()

class MemoryDistiller:

    DISTILL_THRESHOLD = 30       # facts 超过 30 条触发
    MAX_DISTILLED_CHARS = 300    # 蒸馏摘要最大 300 字
    KEEP_TOP_FACTS = 5

    def __init__(self, storage: MemoryStorage, llm_client=None):
        self.storage = storage
        self.llm_client = llm_client

    async def distill_user_space(self, us: UserSpace) -> str:
        if len(us.facts) <= self.DISTILL_THRESHOLD:
            return us.distilled_summary

        core = [f for f in us.facts if f.tier == MemoryTier.CORE_IDENTITY]
        long_term = [f for f in us.facts if f.tier == MemoryTier.LONG_TERM_PREFERENCE]
        temp = [f for f in us.facts if f.tier == MemoryTier.TEMPORARY_CONTEXT]

        # core_identity 不蒸馏，直接保留
        core_text = "\n".join(f"• {f.content}" for f in core)

        # long_term → 蒸馏
        long_text = "\n".join(
            f"[{f.category}] {f.content} (置信度:{f.confidence:.1f})" for f in long_term
        )
        long_summary = ""
        if long_text.strip():
            long_summary = await self._call_llm(
                long_text, self.MAX_DISTILLED_CHARS, "长期偏好"
            )

        # temporary → 更短
        temp_text = "\n".join(f"• {f.content}" for f in temp)
        temp_summary = ""
        if temp_text.strip():
            temp_summary = await self._call_llm(temp_text, 100, "临时上下文")

        # 事件时间线
        events_summary = ""
        if us.events:
            sorted_e = sorted(us.events, key=lambda e: e.get("importance", 0.5), reverse=True)[:10]
            timeline = "\n".join(
                f"[{e.get('timestamp','')[:10]}] {e.get('core_goal','')} — {e.get('final_result','')}"
                for e in sorted_e
            )
            events_summary = await self._call_llm(timeline, 200, "事件时间线")

        # 拼接
        distilled = ""
        if core_text:
            distilled += f"【核心身份】\n{core_text}\n\n"
        if long_summary:
            distilled += f"【长期偏好】\n{long_summary}\n\n"
        if temp_summary:
            distilled += f"【当前上下文】\n{temp_summary}\n\n"
        if events_summary:
            distilled += f"【重要事件】\n{events_summary}"

        us.distilled_summary = distilled
        us.last_distilled_at = datetime.now().isoformat()
        self.storage.save_user_space(us)
        return distilled

    async def distill_agent_memory(self, am: AgentMemory) -> str:

import numpy as np
from ..core.user_space import MemoryFact

class EmbeddingIndex:
    """
    基于 embedding 的语义检索
    推荐本地模型: BAAI/bge-small-zh-v1.5 (512维, 中文好, 速度快)
    或 OpenAI: text-embedding-3-small (1536维)
    """

    def __init__(self, model_name="BAAI/bge-small-zh-v1.5", use_openai=False):
        self.model_name = model_name
        self.use_openai = use_openai
        self.embeddings = {}      # fact_id -> np.ndarray
        self.facts = {}           # fact_id -> MemoryFact
        self._model = None

    def _get_model(self):
        if self._model:
            return self._model
        if self.use_openai:
            import openai
            self._model = openai.OpenAI()
        else:
            from sentence_transformers import SentenceTransformer
            self._model = SentenceTransformer(self.model_name)
        return self._model

    def embed(self, text: str) -> np.ndarray:
        model = self._get_model()
        if self.use_openai:
            resp = model.embeddings.create(input=text, model="text-embedding-3-small")
            return np.array(resp.data[0].embedding, dtype=np.float32)
        else:
            return np.array(model.encode(text, normalize_embeddings=True), dtype=np.float32)

    def index_facts(self, facts: list):
        if not facts:
            return
        texts = [f"{f.category}: {f.content}" for f in facts]
        model = self._get_model()
        if self.use_openai:
            resp = model.embeddings.create(
                input=[t[:8191] for t in texts], model="text-embedding-3-small"
            )
            for i, fact in enumerate(facts):
                self.embeddings[fact.id] = np.array(resp.data[i].embedding, dtype=np.float32)
                self.facts[fact.id] = fact
        else:
            vectors = model.encode(texts, normalize_embeddings=True, batch_size=32)
            for i, fact in enumerate(facts):
                self.embeddings[fact.id] = np.array(vectors[i], dtype=np.float32)
                self.facts[fact.id] = fact

    def search(self, query: str, top_k=8, min_score=0.3) -> list:
        """
        语义搜索，返回 [(MemoryFact, score), ...]
        tier 加权: core_identity x1.5, long_term x1.0, temporary x0.7
        """
        if not self.embeddings:
            return []
        q_emb = self.embed(query)
        tier_weights = {
            "core_identity": 1.5,
            "long_term_preference": 1.0,
            "temporary_context": 0.7,
        }
        results = []
        for fid, emb in self.embeddings.items():
            score = float(np.dot(q_emb, emb))  # cosine (已归一化)
            fact = self.facts[fid]
            weighted = score * tier_weights.get(fact.tier, 1.0)
            if weighted >= min_score:
                results.append((fact, weighted))
        results.sort(key=lambda x: x[1], reverse=True)
        return results[:top_k]

    def remove_fact(self, fact_id: str):
        self.embeddings.pop(fact_id, None)
        self.facts.pop(fact_id, None)

from ..core.user_space import MemoryFact, MemoryLayer
from ..core.storage import MemoryStorage
from .embedding_index import EmbeddingIndex

class MemorySearchEngine:
    """优先语义搜索，回退关键词搜索"""

    def __init__(self, storage: MemoryStorage):
        self.storage = storage
        self.index = EmbeddingIndex()
        self._built = False

    async def rebuild_index(self, agent_id=None):
        all_facts = []
        us = self.storage.load_user_space()
        all_facts.extend(us.facts)
        if agent_id:
            am = self.storage.load_agent_memory(agent_id)
            all_facts.extend(am.facts)
        else:
            for f in self.storage.agents_dir.glob("*.json"):
                am = self.storage.load_agent_memory(f.stem)
                all_facts.extend(am.facts)
        self.index = EmbeddingIndex()
        self.index.index_facts(all_facts)
        self._built = True

    async def search(self, query: str, agent_id: str, top_k=8) -> list:
        if self._built:
            results = self.index.search(query, top_k=top_k)
            # 过滤权限：只返回全局层或本 Agent 的
            return [(f, s) for f, s in results
                    if f.layer == MemoryLayer.USER or f.source_agent_id == agent_id]
        return self._keyword_fallback(query, agent_id, top_k)

    def _keyword_fallback(self, query, agent_id, top_k):
        us = self.storage.load_user_space()
        am = self.storage.load_agent_memory(agent_id)
        q_terms = set(query.lower().split())
        results = []
        for fact in us.facts + am.facts:
            f_terms = set(fact.content.lower().split())
            inter = q_terms & f_terms
            union = q_terms | f_terms
            if union:
                score = len(inter) / len(union)
                if score > 0.1:
                    results.append((fact, score))
        results.sort(key=lambda x: x[1], reverse=True)
        return results[:top_k]

from collections import defaultdict
from datetime import datetime
from ..core.user_space import MemoryFact, MemoryTier

class MemoryEvent:
    """一条记忆变更事件"""
    def __init__(self, event_type: str, agent_id: str, fact: MemoryFact):
        self.event_type = event_type  # "fact_added" | "fact_updated" | "fact_removed"
        self.agent_id = agent_id
        self.fact = fact
        self.timestamp = datetime.now().isoformat()

    def should_broadcast(self) -> bool:
        """只有高价值变更才广播，避免噪音"""
        if self.fact.tier == MemoryTier.CORE_IDENTITY and self.fact.confidence > 0.7:
            return True
        if self.fact.category == "correction":
            return True
        return False

class MemoryEventBus:
    """
    跨 Agent 记忆变更通知。
    Agent A 写入高价值 fact → 事件总线 → 推送到订阅者的 pending 队列
    订阅者下次构建注入上下文时，先消费 pending
    """
    def __init__(self):
        self._subscriptions = defaultdict(list)  # source -> [targets]
        self._pending = defaultdict(list)         # target -> [MemoryEvent]
        self._history = []
        self._max_history = 100

    def subscribe(self, source_agent_id: str, target_agent_id: str):
        if target_agent_id not in self._subscriptions[source_agent_id]:
            self._subscriptions[source_agent_id].append(target_agent_id)

    def unsubscribe(self, source_agent_id: str, target_agent_id: str):
        subs = self._subscriptions.get(source_agent_id, [])
        if target_agent_id in subs:
            subs.remove(target_agent_id)

    def publish(self, event: MemoryEvent):
        self._history.append(event)
        if len(self._history) > self._max_history:
            self._history.pop(0)
        if not event.should_broadcast():
            return
        for target in self._subscriptions.get(event.agent_id, []):
            self._pending[target].append(event)

    def get_pending(self, agent_id: str) -> list:
        events = self._pending.get(agent_id, []).copy()
        self._pending[agent_id] = []
        return events

    def has_pending(self, agent_id: str) -> bool:
        return len(self._pending.get(agent_id, [])) > 0

    def format_pending_for_injection(self, agent_id: str) -> str:
        """格式化为注入文本，在 build_injection_context 时调用"""
        events = self.get_pending(agent_id)
        if not events:
            return ""
        labels = {"fact_added": "新增", "fact_updated": "更新", "fact_removed": "删除"}
        lines = ["【来自其他 Agent 的记忆更新】"]
        for e in events:
            action = labels.get(e.event_type, e.event_type)
            lines.append(f"- [{action}] 来自 {e.agent_id}: {e.fact.content}")
        return "\n".join(lines)

from datetime import datetime
from ..core.user_space import UserSpace, AgentMemory, MemoryFact, MemoryTier

class MemoryExporter:
    def __init__(self, storage: MemoryStorage):
        self.storage = storage

    def export_full_memory(self, agent_id=None) -> str:
        us = self.storage.load_user_space()
        md = self._format_user_space(us)
        if agent_id:
            am = self.storage.load_agent_memory(agent_id)
            md += "\n\n---\n\n" + self._format_agent(am)
        md += f"\n\n---\n\n> 最后更新: {datetime.now().strftime('%Y-%m-%d %H:%M')}"
        return md

    def _format_user_space(self, us: UserSpace) -> str:
        p = us.profile
        lines = ["# 用户记忆档案\n", "## 👤 用户档案\n"]
        lines.append(f"- **姓名**: {p.name or '未设置'}")
        if p.nickname:
            lines.append(f"- **昵称**: {p.nickname}")
        if p.preferred_name and p.preferred_name != p.name:
            lines.append(f"- **称呼**: {p.preferred_name}")
        lines.append(f"- **职业**: {p.occupation or '未设置'}")
        lines.append(f"- **位置**: {p.location or '未设置'}")
        if p.interests:
            lines.append(f"- **兴趣**: {', '.join(p.interests)}")

        c = us.context
        if c.work or c.personal or c.top_of_mind:
            lines.append("\n## 📍 当前上下文\n")
            if c.work:
                lines.append(f"- **工作**: {c.work}")
            if c.personal:
                lines.append(f"- **个人**: {c.personal}")
            if c.top_of_mind:
                lines.append(f"- **近期关注**: {c.top_of_mind}")

        if us.distilled_summary:
            lines.append("\n## 🧠 记忆摘要\n" + us.distilled_summary)

        # 按 tier 分组
        for tier, title in [(MemoryTier.CORE_IDENTITY, "🔑 核心身份"),
                            (MemoryTier.LONG_TERM_PREFERENCE, "📋 长期偏好"),
                            (MemoryTier.TEMPORARY_CONTEXT, "⏳ 临时上下文")]:
            facts = [f for f in us.facts if f.tier == tier]

import re
from datetime import datetime
from ..core.user_space import (
    UserProfile, UserSpace, MemoryFact, MemoryTier
)

class MarkdownParser:
    """用户在 UI 编辑 Markdown 后，解析回结构化记忆"""

    PROFILE_MAP = {
        "姓名": "name", "昵称": "nickname", "称呼": "preferred_name",
        "性别": "gender", "职业": "occupation", "位置": "location",
        "时区": "timezone", "语言": "languages",
        "兴趣": "interests", "爱好": "hobbies",
    }

    def parse_and_update(self, md: str, us: UserSpace) -> UserSpace:
        self._parse_profile(md, us)
        self._parse_context(md, us)
        self._parse_facts(md, us)
        return us

    def _parse_profile(self, md, us):
        match = re.search(r"## 👤 用户档案(.*?)(?=## |\Z)", md, re.DOTALL)
        if not match:
            return
        for line in match.group(1).strip().split("\n"):
            m = re.match(r"- \*\*(.+?)\*\*[:：]\s*(.+)", line.strip())
            if m:
                name, val = m.group(1).strip(), m.group(2).strip()
                attr = self.PROFILE_MAP.get(name)
                if attr:
                    if attr in ("languages", "interests", "hobbies"):
                        setattr(us.profile, attr, [x.strip() for x in val.split(",")])
                    else:
                        setattr(us.profile, attr, val)

    def _parse_context(self, md, us):
        match = re.search(r"## 📍 当前上下文(.*?)(?=## |\Z)", md, re.DOTALL)
        if not match:
            return
        for line in match.group(1).strip().split("\n"):
            m = re.match(r"- \*\*(.+?)\*\*[:：]\s*(.+)", line.strip())
            if m:
                name, val = m.group(1).strip(), m.group(2).strip()
                if name == "工作":
                    us.context.work = val
                elif name == "个人":
                    us.context.personal = val
                elif name == "近期关注":
                    us.context.top_of_mind = val

    def _parse_facts(self, md, us):
        tier_sections = [
            ("🔑 核心身份", MemoryTier.CORE_IDENTITY),
            ("📋 长期偏好", MemoryTier.LONG_TERM_PREFERENCE),
            ("⏳ 临时上下文", MemoryTier.TEMPORARY_CONTEXT),
        ]
        emoji_cat = {"💛": "preference", "📚": "knowledge", "📍": "context",
                     "🔄": "behavior", "🎯": "goal", "⚠️": "correction"}

        for title, tier in tier_sections:
            match = re.search(rf"## {re.escape(title)}(.*?)(?=## |\Z)", md, re.DOTALL)
            if not match:
                continue
            existing = {f.content for f in us.facts if f.tier == tier}
            for line in match.group(1).strip().split("\n"):
                if not line.strip().startswith("-"):
                    continue
                cm = re.match(r"- ([^\s])\s+(.+)", line.strip())
                if cm:
                    content = cm.group(2).strip()
                    if content not in existing:
                        us.facts.append(MemoryFact(
                            id=f"fact_manual_{len(us.facts)}",
                            content=content,
                            category=emoji_cat.get(cm.group(1), "knowledge"),
                            tier=tier,
                            layer="user",
                            confidence=1.0,
                            created_at=datetime.now().isoformat(),
                            source_agent_id="manual_edit",
                        ))
                        existing.add(content)

                sections.append("【相关记忆】\n" + "\n".join(lines))

        # P5: 当前线程工作记忆
        if thread_id and thread_id in am.working_memory:

            if t.get("key_context"):

            if t.get("recent_messages_summary"):

            if tp:
                sections.append("【当前对话】\n" + "\n".join(tp))

        if bus:
            sections.append(bus)

        full = "\n\n".join(sections)
        est = len(full) // 2
        if est > token_budget:
            full = "\n\n".join(full.split("\n\n")[:3])
            full += f"\n\n[记忆精简: {est}t > 预算 {token_budget}t]"
        return full

def migrate_v2_to_v3(memory_root="memory"):
    """运行一次。把旧混合记忆拆分为 UserSpace + AgentMemory"""
    storage = MemoryStorage(memory_root)
    root = Path(memory_root)

    agent_dirs = [d for d in root.iterdir()
                  if d.is_dir() and d.name.startswith("agent_")]
    if not agent_dirs:
        print("没有旧格式数据")
        return

    user_facts = []
    promote_kw = [
        "姓名", "名字", "叫", "昵称", "职业", "工作", "公司",
        "位置", "城市", "住在", "语言", "母语",
    ]
    best_profile = {}

    for ad in agent_dirs:
        old_file = ad / "memory.json"
        if not old_file.exists():
            continue
        old = json.loads(old_file.read_text(encoding="utf-8"))
        aid = ad.name
        new_facts = []
        p = old.get("profile", {})
        if len(p) > len(best_profile):
            best_profile = p

        for fd in old.get("facts", []):
            fact = MemoryFact(
                id=fd.get("id", ""), content=fd.get("content", ""),
                category=fd.get("category", "preference"),
                tier=fd.get("tier", "long_term_preference"),
                layer="agent", confidence=fd.get("confidence", 0.7),
                source_agent_id=aid,
            )
            promote = (
                fact.tier == MemoryTier.CORE_IDENTITY
                or (fact.category == "knowledge" and fact.confidence > 0.8)
                or fact.category == "correction"
                or any(kw in fact.content for kw in promote_kw)
            )
            if promote:
                fact.layer = MemoryLayer.USER
                user_facts.append(fact)
            else:
                new_facts.append(fact)

        storage._atomic_write_json(
            storage.agents_dir / f"{aid}.json",
            {"agent_id": aid,
             "facts": [storage._to_dict(f) for f in new_facts],
             "events": old.get("episodic_events", []),
             "working_memory": old.get("working_memory", {}),
             "domain_summary": old.get("domain_summary", ""),
             "version": 1}
        )

    from .user_space import UserSpace
    us = UserSpace()
    for k, v in best_profile.items():
        if hasattr(us.profile, k):
            setattr(us.profile, k, v)
    us.facts = user_facts
    storage.save_user_space(us)

    # 备份旧数据
    bk = root / "v2_backup"
    bk.mkdir(exist_ok=True)
    for d in agent_dirs:
        f = d / "memory.json"
        if f.exists():
            shutil.copy2(f, bk / f"{d.name}_memory.json")
    sr = root / "sharing_rules.json"
    if sr.exists():
        shutil.copy2(sr, bk / "sharing_rules.json")

    print(f"迁移完成！UserSpace: {len(user_facts)} 条全局事实")
    print(f"旧数据备份: {bk}")
