### 2. 修改重写接口
在 `chat.py` 中更新重写逻辑：
```python
@router.post("/conversations/{conv_id}/messages/{msg_id}/rewrite")
async def rewrite_message(conv_id: str, msg_id: str, request: RewriteRequest):
    conv_store = ConversationStore()
    conversation = conv_store.get(conv_id)
    
    # 找到要重写的消息
    for msg in conversation["messages"]:
        if msg["id"] == msg_id and msg["role"] == "assistant":
            # 如果是第一次重写，初始化versions数组
            if "versions" not in msg or len(msg["versions"]) == 0:
                # 把原有内容作为第一个版本
                msg["versions"] = [{
                    "id": str(uuid.uuid4()),
                    "content": msg["content"],
                    "suggestedQuestions": msg.get("suggestedQuestions"),
                    "created_at": msg["created_at"],
                    "model": msg.get("model"),
                    "provider": msg.get("provider")
                }]
            
            # 生成新版本
            new_version = {
                "id": str(uuid.uuid4()),
                "content": "",  # 流式填充
                "suggestedQuestions": None,
                "created_at": datetime.now().isoformat(),
                "model": request.model,
                "provider": request.provider
            }
            
            # 添加到版本列表
            msg["versions"].append(new_version)
            # 设置为当前版本
            msg["current_version"] = len(msg["versions"]) - 1
            
            # 保存对话
            conv_store.set(conversation)
            
            # 返回新版本ID，开始流式输出
            return {
                "success": True,
                "version_id": new_version["id"],
                "version_index": len(msg["versions"]) - 1
            }
    
    raise HTTPException(status_code=404, detail="消息不存在")
```

### 4. 更新Chat Store
在 `chat.ts` 中添加版本管理方法：
```ts
// 更新当前版本
function updateMessageVersion(messageId: string, versionIndex: number) {
  const messages = convMessages.value[activeConvId.value]
  const msg = messages.find(m => m.id === messageId)
  
  if (msg && msg.versions && versionIndex < msg.versions.length) {
    msg.current_version = versionIndex
    // 更新content字段，兼容旧逻辑
    msg.content = msg.versions[versionIndex].content
    msg.suggestedQuestions = msg.versions[versionIndex].suggestedQuestions
  }
}

// 删除版本
function deleteMessageVersion(messageId: string, versionIndex: number) {
  const messages = convMessages.value[activeConvId.value]
  const msg = messages.find(m => m.id === messageId)
  
  if (msg && msg.versions && msg.versions.length > 1) {
    msg.versions.splice(versionIndex, 1)
    // 如果删除的是当前版本，切换到前一个
    if (msg.current_version >= versionIndex) {
      msg.current_version = Math.max(0, msg.current_version - 1)
    }
    // 更新content字段
    msg.content = msg.versions[msg.current_version].content
    msg.suggestedQuestions = msg.versions[msg.current_version].suggestedQuestions
    
    // 保存到后端
    api.post(`/api/v1/chat/conversations/${activeConvId.value}/messages/${messageId}/versions`, {
      versions: msg.versions,
      current_version: msg.current_version
    })
  }
}

// 重写消息
async function rewriteMessage(messageId: string) {
```
